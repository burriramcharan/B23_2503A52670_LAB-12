def linear_search(arr, target):
	"""Return the target's index, or -1 if it is absent.

	Best-case time: O(1); average-case: O(n); worst-case: O(n).
	"""
	for index, value in enumerate(arr):
		if value == target:
			return index
	return -1

def binary_search(arr, target):
	"""Return target's index in sorted arr, or -1 if it is absent.

	Time complexity: best O(1), average O(log n), worst O(log n).

	The list must be sorted so comparisons indicate which half can be
	discarded. Binary Search is faster than Linear Search on large sorted
	lists because it halves the search range each step; Linear Search checks
	elements one at a time.
	"""
	left = 0
	right = len(arr) - 1

	while left <= right:
		middle = (left + right) // 2
		if arr[middle] == target:
			return middle
		if arr[middle] < target:
			left = middle + 1
		else:
			right = middle - 1

	return -1

from __future__ import annotations


def factorial(n: int) -> int:
    """Return n! for a non-negative integer."""
    if n < 0:
        raise ValueError("n must be non-negative")

    result = 1
    for value in range(2, n + 1):
        result *= value
    return result


def is_prime(value: int) -> bool:
    """Return True if value is prime, otherwise False."""
    if value < 2:
        return False
    if value in (2, 3):
        return True
    if value % 2 == 0:
        return False

    limit = int(value ** 0.5)
    for divisor in range(3, limit + 1, 2):
        if value % divisor == 0:
            return False
    return True


def average(numbers):
    """Return the arithmetic mean of a non-empty sequence of numbers."""
    if not numbers:
        raise ValueError("numbers cannot be empty")
    return sum(numbers) / len(numbers)


def count_vowels(text: str) -> int:
    """Count vowels in a string, ignoring case."""
    vowels = "aeiou"
    return sum(1 for ch in text.lower() if ch in vowels)


def celsius_to_fahrenheit(celsius: float) -> float:
    """Convert Celsius to Fahrenheit."""
    return (celsius * 9 / 5) + 32


def fahrenheit_to_celsius(fahrenheit: float) -> float:
    """Convert Fahrenheit to Celsius."""
    return (fahrenheit - 32) * 5 / 9


def main() -> None:
    """Simple demonstration of the utility functions."""
    print("Lab 12 Utility Demo")
    print("Factorial of 5:", factorial(5))
    print("Is 17 prime?", is_prime(17))
    print("Average of [2, 4, 6, 8]:", average([2, 4, 6, 8]))
    print("Vowels in 'Hello, World!':", count_vowels("Hello, World!"))
    print("32C in Fahrenheit:", celsius_to_fahrenheit(32))
    print("68F in Celsius:", fahrenheit_to_celsius(68))


if __name__ == "__main__":
    main()

""Compare linear and binary search for product IDs."""

from time import perf_counter


def linear_search(product_ids, target):
	comparisons = 0
	for index, product_id in enumerate(product_ids):
		comparisons += 1
		if product_id == target:
			return index, comparisons
	return -1, comparisons


def binary_search(product_ids, target):
	low, high = 0, len(product_ids) - 1
	comparisons = 0
	while low <= high:
		middle = (low + high) // 2
		comparisons += 1
		if product_ids[middle] == target:
			return middle, comparisons
		comparisons += 1
		if product_ids[middle] < target:
			low = middle + 1
		else:
			high = middle - 1
	return -1, comparisons


def timed_search(search, product_ids, target, repeats=200):
	"""Return average runtime in microseconds and comparisons per search."""
	_, comparisons = search(product_ids, target)
	start = perf_counter()
	for _ in range(repeats):
		search(product_ids, target)
	elapsed_us = (perf_counter() - start) * 1_000_000 / repeats
	return elapsed_us, comparisons


def main():
	sizes = (10, 100, 1_000)
	print(f"{'Size':>6} {'Case':>9} {'Algorithm':<9} {'Result':>10} "
		  f"{'Comparisons':>11} {'Avg time (us)':>14}")
	print("-" * 66)

	for size in sizes:
		# Binary search requires sorted product IDs.
		product_ids = list(range(100_000, 100_000 + size))
		cases = (
			("first", product_ids[0]),
			("middle", product_ids[size // 2]),
			("last", product_ids[-1]),
			("missing", product_ids[-1] + 1),
		)
		for label, target in cases:
			for name, search in (("Linear", linear_search),
								 ("Binary", binary_search)):
				index, _ = search(product_ids, target)
				elapsed_us, comparisons = timed_search(search, product_ids, target)
				result = str(index) if index >= 0 else "not found"
				print(f"{size:6} {label:>9} {name:<9} {result:>10} "
					  f"{comparisons:11} {elapsed_us:14.3f}")

	print("\nTime complexity and limitations:")
	print("- Linear search takes O(n) time in the average and worst cases and")
	print("  O(1) extra space. It works on unsorted data, but can inspect every ID.")
	print("- Binary search takes O(log n) time and O(1) extra space, but requires")
	print("  sorted data and efficient random access. Sorting first costs O(n log n).")
	print("Conclusion: binary search is generally better for already-sorted product")
	print("IDs, especially as the collection grows, because it needs fewer comparisons.")


if __name__ == "__main__":
	main()
